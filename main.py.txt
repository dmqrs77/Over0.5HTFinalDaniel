# ========================= CONFIG =========================
TOKEN = os.getenv("TOKEN")  # ← variável de ambiente do Telegram
API_KEY = os.getenv("API_KEY")  # ← variável de ambiente da API-Football
API_FOOTBALL = "https://v3.football.api-sports.io"
HEADERS = {"x-apisports-key": API_KEY}

# Filtros V4
MIN_XG_MIN = 0.08
POSSE_OSCILACAO = 15
MIN_ALERTAS_DIA = 5
JANELA_EARLY = (15, 20)
JANELA_CLUTCH = (25, 30)
ODDS_RANGE = (1.50, 2.20)
LIGAS_PREMIUM = [34, 39, 40, 61, 78, 88, 103, 135, 144, 253]

BANCA_PADRAO = 500
PERCENT_ENTRADA = 0.0202

# ======================== ESTADO GLOBAL ========================
estado = {
    "banca": BANCA_PADRAO,
    "alertas_hoje": [],
    "jogos_monitorados": {},
    "usuarios": {}
}

# ========================= LOGGING =========================
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(_name_)

# ======================= FUNÇÕES AUX =======================
def calcular_entrada() -> float:
    return round(estado["banca"] * PERCENT_ENTRADA, 2)

def formatar_alerta(jogo: Dict, tipo: str = "EARLY") -> str:
    odd = jogo.get("odd_live", 1.75)
    xg = jogo.get("xg_live", 0.0)
    chutes = jogo.get("chutes", "0-0")
    escanteios = jogo.get("escanteios", "0-0")
    minuto = jogo.get("minuto", 15)

    entrada = calcular_entrada()
    banca_str = f"R${estado['banca']:,.0f}".replace(",", ".")

    grafico_xg = "█" * min(int(xg * 10), 10) + "░" * (10 - min(int(xg * 10), 10))
    grafico_pressao = "🔥" if jogo.get("pressao_alta", False) else "⚪"

    msg = f"""
⚡ {tipo} BIRD ({minuto}')
{jogo['home']} × {jogo['away']}
Odd: {odd:.2f} | XG: {xg:.2f} {grafico_xg}
Chutes: {chutes} | Escanteios: {escanteios}
Pressão: {grafico_pressao}
Entrada: R${entrada} (2% da banca {banca_str})
"""
    return msg.strip()

# ======================= API FOOTBALL =======================
async def get_fixtures_live(session) -> List[Dict]:
    url = f"{API_FOOTBALL}/fixtures?live=all"
    async with session.get(url, headers=HEADERS) as resp:
        if resp.status != 200:
            return []
        data = await resp.json()
        return data.get("response", [])

async def get_fixture_stats(session, fixture_id: int) -> Optional[List[Dict]]:
    url = f"{API_FOOTBALL}/fixtures/statistics?fixture={fixture_id}"
    async with session.get(url, headers=HEADERS) as resp:
        if resp.status != 200:
            return None
        data = await resp.json()
        return data.get("response", [{}, {}])

# ======================= ANÁLISE V4 =======================
def analisar_jogo_v4(jogo: Dict, stats: List[Dict]) -> Optional[Dict]:
    minuto = jogo["fixture"]["status"]["elapsed"] or 0
    if minuto < 15 or minuto > 30:
        return None

    xg_home = float(stats[0].get("statistics", [{}])[0].get("value", 0)) if stats[0].get("statistics") else 0
    xg_away = float(stats[1].get("statistics", [{}])[0].get("value", 0)) if stats[1].get("statistics") else 0
    xg_total = xg_home + xg_away
    xg_min = xg_total / max(minuto, 1)

    chutes_home = stats[0].get("statistics", [{}])[5].get("value", 0) if len(stats[0].get("statistics", [])) > 5 else 0
    chutes_away = stats[1].get("statistics", [{}])[5].get("value", 0) if len(stats[1].get("statistics", [])) > 5 else 0
    escanteios_home = stats[0].get("statistics", [{}])[9].get("value", 0) if len(stats[0].get("statistics", [])) > 9 else 0
    escanteios_away = stats[1].get("statistics", [{}])[9].get("value", 0) if len(stats[1].get("statistics", [])) > 9 else 0

    posse_home = int(stats[0].get("statistics", [{}])[10].get("value", "0%").strip("%") or 0)
    posse_away = int(stats[1].get("statistics", [{}])[10].get("value", "0%").strip("%") or 0)
    posse_diff = abs(posse_home - posse_away)

    pressao_alta = xg_min > MIN_XG_MIN and posse_diff > POSSE_OSCILACAO

    odd = 1.75
    if jogo.get("odds"):
        for odd_data in jogo["odds"]:
            if odd_data["name"] == "Over 0.5" and "1st Half" in odd_data["values"][0]["value"]:
                odd = float(odd_data["values"][0]["odd"])
                break

    if odd > 2.00:
        if xg_home < 1.2:
            return None

    if not (ODDS_RANGE[0] <= odd <= ODDS_RANGE[1]):
        return None

    return {
        "id": jogo["fixture"]["id"],
        "home": jogo["teams"]["home"]["name"],
        "away": jogo["teams"]["away"]["name"],
        "minuto": minuto,
        "xg_live": round(xg_total, 2),
        "xg_min": round(xg_min, 3),
        "chutes": f"{chutes_home}-{chutes_away}",
        "escanteios": f"{escanteios_home}-{escanteios_away}",
        "pressao_alta": pressao_alta,
        "odd_live": round(odd, 2),
        "liga_id": jogo["league"]["id"]
    }

# ======================= MONITORAMENTO =======================
async def monitorar_jogos(application: Application):
    async with aiohttp.ClientSession() as session:
        while True:
            try:
                fixtures = await get_fixtures_live(session)
                candidatos = []

                for jogo in fixtures:
                    if jogo["league"]["id"] not in LIGAS_PREMIUM:
                        continue

                    stats = await get_fixture_stats(session, jogo["fixture"]["id"])
                    if not stats or len(stats) < 2:
                        continue

                    analise = analisar_jogo_v4(jogo, stats)
                    if not analise:
                        continue

                    jogo_id = analise["id"]
                    minuto = analise["minuto"]

                    if jogo_id in estado["jogos_monitorados"]:
                        continue

                    candidatos.append((analise, minuto))

                candidatos.sort(key=lambda x: x[0]["xg_min"], reverse=True)

                alertas_hoje = len(estado["alertas_hoje"])
                hora_atual = datetime.now().hour
                if alertas_hoje < MIN_ALERTAS_DIA and hora_atual in [12, 15, 18, 21]:
                    if candidatos:
                        analise, _ = candidatos[0]
                        await enviar_alerta(application, analise, "FORCED")
                        estado["jogos_monitorados"][analise["id"]] = True
                        continue

                for analise, minuto in candidatos:
                    jogo_id = analise["id"]

                    if JANELA_EARLY[0] <= minuto <= JANELA_EARLY[1]:
                        await enviar_alerta(application, analise, "EARLY")
                        estado["jogos_monitorados"][jogo_id] = {"enviado_early": True}

                    elif JANELA_CLUTCH[0] <= minuto <= JANELA_CLUTCH[1]:
                        if estado["jogos_monitorados"].get(jogo_id, {}).get("enviado_early"):
                            await enviar_alerta(application, analise, "CLUTCH")
                            estado["jogos_monitorados"][jogo_id]["enviado_clutch"] = True

            except Exception as e:
                logger.error(f"Erro no loop: {e}")

            await asyncio.sleep(60)

async def enviar_alerta(application: Application, jogo: Dict, tipo: str):
    msg = formatar_alerta(jogo, tipo)
    keyboard = [[InlineKeyboardButton("📊 Stats", callback_data=f"stats_{jogo['id']}")]]
    reply_markup = InlineKeyboardMarkup(keyboard)

    for user_id in estado["usuarios"]:
        try:
            await application.bot.send_message(
                chat_id=user_id,
                text=msg,
                parse_mode="Markdown",
                reply_markup=reply_markup
            )
        except:
            pass

    estado["alertas_hoje"].append({**jogo, "tipo": tipo, "hora": datetime.now().isoformat()})

# ======================= COMANDOS =======================
async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    estado["usuarios"][user_id] = True

    await update.message.reply_text(
        "🚀 OVER 0.5 HT DANIEL – V4 CLEAN MODE\n\n"
        "✅ 5+ alertas/dia GARANTIDO\n"
        "✅ Entrada fixa: 2% da banca\n"
        "✅ Análise: XG + chutes + escanteios + pressão\n"
        "✅ Janela: 15-30' do 1T\n\n"
        "Comandos: /status | /hoje | /banca 1000",
        parse_mode="Markdown"
    )

async def status(update: Update, context: ContextTypes.DEFAULT_TYPE):
    jogos_ativos = len(estado["jogos_monitorados"])
    alertas = len(estado["alertas_hoje"])

    await update.message.reply_text(
        f"STATUS AO VIVO\n"
        f"Usuários ativos: {len(estado['usuarios'])}\n"
        f"Alertas hoje: {alertas}\n"
        f"Jogos monitorados: {jogos_ativos}\n"
        f"Entrada fixa: 2%\n\n"
        f"Nenhum jogo ao vivo no momento" if jogos_ativos == 0 else "🔥 Jogos rolando!",
        parse_mode="Markdown"
    )

async def banca(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not context.args:
        await update.message.reply_text("Uso: /banca 500")
        return
    try:
        valor = float(context.args[0])
        estado["banca"] = valor
        await update.message.reply_text(f"Banca atualizada: R${valor:,.0f}".replace(",", "."), parse_mode="Markdown")
    except:
        await update.message.reply_text("Valor inválido.")

async def hoje(update: Update, context: ContextTypes.DEFAULT_TYPE):
    hoje = [a for a in estado["alertas_hoje"] if datetime.fromisoformat(a["hora"]).date() == datetime.now().date()]
    lucro = sum([calcular_entrada() * (a["odd_live"] - 1) for a in hoje])

    await update.message.reply_text(
        f"📊 RESUMO DO DIA\n"
        f"Alertas: {len(hoje)}\n"
        f"Entrada: R${calcular_entrada()}/alerta\n"
        f"Lucro estimado: +R${lucro:.0f}\n"
        f"ROI: {(lucro/estado['banca'])*100:.1f}%\n",
        parse_mode="Markdown"
    )

async def button_callback(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    if query.data.startswith("stats_"):
        await query.edit_message_text("📈 Stats detalhadas em desenvolvimento...")

# ======================= MAIN =======================
def main():
    app = Application.builder().token(TOKEN).build()

    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("status", status))
    app.add_handler(CommandHandler("banca", banca))
    app.add_handler(CommandHandler("hoje", hoje))
    app.add_handler(CallbackQueryHandler(button_callback))

    loop = asyncio.get_event_loop()
    loop.create_task(monitorar_jogos(app))

    print("🤖 V4 CLEAN MODE RODANDO – Render Cloud Active")
    app.run_polling()

if _name_ == "_main_":
    main()
